#include <stdio.h>
#include <string.h>
void reverse(char* str){
    int len = strlen(str);
    char temp;
    char* p1 = str;
    char* p2 = str + len - 1;
    while(p1 < p2){
        temp = *p1; *p1 = *p2; *p2 = temp;
        p1++; p2--;
    }
}
int main(int argc, char* argv[]){

    reverse(str);

        printf("%c", str[i]);

    return 0;
}

#include <stdio.h>
typedef struct{ int accNum; double bal; }BankAcc;
double sim_pow(double base, int year){
    int i; double r = 1.0;
    for(i=0; i<year; i++) r = r * base;
    return r;
}
void initAcc(BankAcc *acc, int x, double y){
    acc->accNum = x; acc->bal = y;
}
void xxx(BankAcc *acc, double *en){
    if (*en > 0 && *en < acc->bal) acc->bal = acc->bal - *en;
    else acc->bal = acc->bal + *en;
}
void yyy(BankAcc *acc){
    acc->bal = acc->bal * sim_pow((1+0.1), 3);
}
int main(){
    BankAcc myAcc;
    initAcc(&myAcc, 9981, 2200.0);
    double amount = 100.0;
    xxx(&myAcc, &amount);
    yyy(&myAcc);
    printf("%d and %.2f", myAcc.accNum, myAcc.bal);
    return 0;
}
